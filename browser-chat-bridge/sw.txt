300:async function handleWebSocketMessage(
301:    message
302:) {
303:
304:    try {
305:
306:        // --------------------------------------------------------
307:        // bridge.register response
308:        // --------------------------------------------------------
309:
310:        if (
311:            pendingBridgeRegisterId &&
312:            message.type === "response" &&
313:            message.id ===
314:            pendingBridgeRegisterId
315:        ) {
316:
317:            const accepted =
318:                message.result?.accepted !== false;
319:
320:
321:            if (!accepted) {
322:
323:                bridgeStatus =
324:                    BRIDGE_STATUS.ERROR;
325:
326:
327:                console.error(
328:                    "[ServiceWorker] Bridge registration rejected:",
329:                    message.result
330:                );
331:
332:
333:                return;
334:            }
335:
336:
337:            pendingBridgeRegisterId =
338:                null;
339:
340:
341:            bridgeStatus =
342:                BRIDGE_STATUS.READY;
343:
344:
345:            const remoteHeartbeatInterval =
346:                message.result?.heartbeatInterval;
347:
348:
349:            if (
350:                Number.isFinite(
351:                    remoteHeartbeatInterval
352:                ) &&
353:                remoteHeartbeatInterval >= 1000
354:            ) {
355:
356:                heartbeatInterval =
357:                    remoteHeartbeatInterval;
358:
359:
360:                heartbeat.setIntervalMs(
361:                    remoteHeartbeatInterval
362:                );
363:            }
364:
365:
366:            heartbeat.start();
367:
368:
369:            console.log(
370:                "[ServiceWorker] Bridge ready:",
371:                bridgeId
372:            );
373:
374:
375:            /*
376:             * Re-register all browser agents.
377:             *
378:             * Important after:
379:             * - WebSocket reconnect
380:             * - 9Router restart
381:             * - service worker restart
382:             */
383:            await announceAllAgents();
384:
385:
386:            return;
387:        }
388:
389:
390:        // --------------------------------------------------------
391:        // bridge.register error
392:        // --------------------------------------------------------
393:
394:        if (
395:            pendingBridgeRegisterId &&
396:            message.type === "error" &&
397:            message.id ===
398:            pendingBridgeRegisterId
399:        ) {
400:
401:            bridgeStatus =
402:                BRIDGE_STATUS.ERROR;
403:
404:
405:            pendingBridgeRegisterId =
406:                null;
407:
408:
409:            heartbeat.stop();
410:
411:
412:            console.error(
413:                "[ServiceWorker] Bridge registration error:",
414:                message.error
415:            );
416:
417:
418:            return;
419:        }
420:
421:
422:        // --------------------------------------------------------
423:        // Normal protocol message
424:        // --------------------------------------------------------
425:
426:        await requestRouter.handle(
427:            message
428:        );
429:
430:    } catch (error) {
431:
432:        console.error(
433:            "[ServiceWorker] Failed to process WebSocket message:",
434:            error
435:        );
436:    }
437:}

587:async function handleRuntimeMessage(
588:    message,
589:    sender
590:) {
591:
592:    if (
593:        !message ||
594:        typeof message !== "object"
595:    ) {
596:
597:        throw new Error(
598:            "Invalid runtime message"
599:        );
600:    }
601:
602:
603:    switch (message.type) {
604:
605:        // --------------------------------------------------------
606:        // Popup: bridge status
607:        // --------------------------------------------------------
608:
609:        case INTERNAL_MESSAGE.GET_BRIDGE_STATUS:
610:
611:            return {
612:                bridge: {
613:                    bridgeId,
614:                    wsUrl,
615:                    status:
616:                        bridgeStatus,
617:
618:                    connected:
619:                        websocket.isConnected(),
620:
621:                    heartbeat:
622:                        heartbeat.getStatus()
623:                }
624:            };
625:
626:
627:        // --------------------------------------------------------
628:        // Popup: current agent
629:        // --------------------------------------------------------
630:
631:        case INTERNAL_MESSAGE.GET_CURRENT_AGENT: {
632:
633:            const tabId =
634:                message.tabId ??
635:                sender.tab?.id;
636:
637:
638:            if (!Number.isInteger(tabId)) {
639:
640:                return {
641:                    agent: null
642:                };
643:            }
644:
645:
646:            return {
647:                agent:
648:                    agentRegistry.getByTabId(
649:                        tabId
650:                    )
651:            };
652:        }
653:
654:        case INTERNAL_MESSAGE.GET_AGENTS:
655:            return handleGetAgents();
656:        // --------------------------------------------------------
657:        // Popup: register agent
658:        // --------------------------------------------------------
659:
660:        case INTERNAL_MESSAGE.REGISTER_AGENT:
661:
662:            return await handleRegisterAgent(
663:                message
664:            );
665:
666:
667:        // --------------------------------------------------------
668:        // Popup: unregister agent
669:        // --------------------------------------------------------
670:
671:        case INTERNAL_MESSAGE.UNREGISTER_AGENT:
672:
673:            return await handleUnregisterAgent(
674:                message
675:            );
676:
677:
678:        // --------------------------------------------------------
679:        // Content: provider detected
680:        // --------------------------------------------------------
681:
682:        case INTERNAL_MESSAGE.PROVIDER_DETECTED:
683:
684:            return await handleProviderDetected(
685:                message,
686:                sender
687:            );
688:
689:
690:        // --------------------------------------------------------
691:        // Content: chat started
692:        // --------------------------------------------------------
693:
694:        case INTERNAL_MESSAGE.CHAT_STARTED:
695:
696:            await requestRouter.handleChatStarted({
697:                requestId:
698:                    message.requestId,
699:
700:                agentId:
701:                    message.agentId
702:            });
703:
704:            return {};
705:
706:
707:        // --------------------------------------------------------
708:        // Content: streaming delta
709:        // --------------------------------------------------------
710:
711:        case INTERNAL_MESSAGE.CHAT_DELTA:
712:
713:            requestRouter.handleChatDelta({
714:                requestId:
715:                    message.requestId,
716:
717:                agentId:
718:                    message.agentId,
719:
720:                seq:
721:                    message.seq,
722:
723:                delta:
724:                    message.delta
725:            });
726:
727:            return {};
728:
729:
730:        // --------------------------------------------------------
731:        // Content: completed
732:        // --------------------------------------------------------
733:
734:        case INTERNAL_MESSAGE.CHAT_COMPLETED:
735:
736:            await requestRouter.handleChatCompleted({
737:                requestId:
738:                    message.requestId,
739:
740:                agentId:
741:                    message.agentId,
742:
743:                content:
744:                    message.content,
745:
746:                conversationId:
747:                    message.conversationId ??
748:                    null,
749:
750:                durationMs:
751:                    message.durationMs ??
752:                    null
753:            });
754:
755:            return {};
756:
757:
758:        // --------------------------------------------------------
759:        // Content: error
760:        // --------------------------------------------------------
761:
762:        case INTERNAL_MESSAGE.CHAT_ERROR:
763:
764:            await requestRouter.handleChatError({
765:                requestId:
766:                    message.requestId,
767:
768:                agentId:
769:                    message.agentId,
770:
771:                error:
772:                    message.error
773:            });
774:
775:            return {};
776:
777:
778:        // --------------------------------------------------------
779:        // Content: provider status
780:        // --------------------------------------------------------
781:
782:        case INTERNAL_MESSAGE.PROVIDER_STATUS:
783:
784:            return await handleProviderStatus(
785:                message,
786:                sender
787:            );
788:
789:
790:        default:
791:
792:            throw new Error(
793:                `Unsupported runtime message: ${message.type}`
794:            );
795:    }
796:}

809:async function handleRegisterAgent(
810:    message
811:) {
812:
813:    const agentId =
814:        message.agentId?.trim();
815:
816:    const tabId =
817:        message.tabId;
818:
819:
820:    if (!agentId) {
821:
822:        throw new Error(
823:            "agentId is required"
824:        );
825:    }
826:
827:
828:    if (!Number.isInteger(tabId)) {
829:
830:        throw new Error(
831:            "tabId is required"
832:        );
833:    }
834:
835:
836:    // ----------------------------------------------------------
837:    // Verify tab
838:    // ----------------------------------------------------------
839:
840:    const tab =
841:        await chrome.tabs.get(
842:            tabId
843:        );
844:
845:
846:    if (!tab) {
847:
848:        throw new Error(
849:            `Tab not found: ${tabId}`
850:        );
851:    }
852:
853:
854:    // ----------------------------------------------------------
855:    // Only ChatGPT for MVP
856:    // ----------------------------------------------------------
857:
858:    if (
859:        !tab.url ||
860:        !tab.url.startsWith(
861:            "https://chatgpt.com/"
862:        )
863:    ) {
864:
865:        throw new Error(
866:            "Current tab is not a supported ChatGPT page"
867:        );
868:    }
869:
870:
871:    // ----------------------------------------------------------
872:    // Ask content script for provider information
873:    // ----------------------------------------------------------
874:
875:    let providerInfo = {};
876:
877:
878:    try {
879:
880:        providerInfo =
881:            await chrome.tabs.sendMessage(
882:                tabId,
883:                {
884:                    type:
885:                        INTERNAL_MESSAGE.PROVIDER_STATUS
886:                }
887:            ) || {};
888:
889:    } catch (error) {
890:
891:        console.warn(
892:            "[ServiceWorker] Unable to query content script:",
893:            error
894:        );
895:    }
896:
897:
898:    const agent =
899:        await agentRegistry.register({
900:            agentId,
901:
902:            provider:
903:                PROVIDER.CHATGPT,
904:
905:            tabId,
906:
907:            conversationId:
908:                providerInfo.conversationId ??
909:                null,
910:
911:            title:
912:                providerInfo.title ??
913:                tab.title ??
914:                null,
915:
916:            status:
917:                AGENT_STATUS.IDLE,
918:
919:            capabilities: [
920:                CAPABILITY.TEXT,
921:                CAPABILITY.STREAM,
922:                CAPABILITY.CANCEL
923:            ]
924:        });
925:
926:
927:    // ----------------------------------------------------------
928:    // Notify 9Router
929:    // ----------------------------------------------------------
930:
931:    if (
932:        bridgeStatus ===
933:        BRIDGE_STATUS.READY
934:    ) {
935:
936:        websocket.send(
937:            createAgentRegister({
938:                bridgeId,
939:                agentId:
940:                    agent.agentId,
941:
942:                provider:
943:                    agent.provider,
944:
945:                tabId:
946:                    agent.tabId,
947:
948:                conversationId:
949:                    agent.conversationId,
950:
951:                title:
952:                    agent.title,
953:
954:                status:
955:                    agent.status,
956:
957:                capabilities:
958:                    agent.capabilities
959:            })
960:        );
961:    }
962:
963:
964:    console.log(
965:        "[ServiceWorker] Agent registered:",
966:        agentId
967:    );

971:        agent
972:    };

1058:     *

1065:        );

1069:
1070:        return {
1071:            registered: false
1072:        };
1073:    }
1074:
1075:
1076:    const updated =
1077:        await agentRegistry.update(
1078:            agent.agentId,
1079:            {
1080:                provider:
1081:                    message.provider ||
1082:                    agent.provider,
1083:
1084:                conversationId:
1085:                    message.conversationId ??
1086:                    agent.conversationId,
1087:
1088:                title:
1089:                    message.title ??
1090:                    agent.title,
1091:
